
#include <iostream>
using namespace std;

int main() {
    int o1, o2, o3, o4, o5, n = 0;
    cin >> o1 >> o2 >> o3 >> o4 >> o5;

    double srednia = (o1 + o2 + o3 + o4 + o5) / 5.0;

    if (o1 == 1) n++;
    if (o2 == 1) n++;
    if (o3 == 1) n++;
    if (o4 == 1) n++;
    if (o5 == 1) n++;

    cout << "Srednia: " << srednia << endl;
    cout << "Niedostateczne: " << n;

    return 0;
}
